cmake_minimum_required(VERSION 3.22)
project(binux_native LANGUAGES C)
set(CMAKE_C_STANDARD 11)
if(ANDROID)
 add_library(binux_process SHARED process.c)
 target_link_options(binux_process PRIVATE -Wl,-z,max-page-size=16384)
 # APK executable entries must be named lib*.so; these are PIE executables,
 # not libraries loaded using dlopen. No writable-data executable downloads.
 add_executable(binuxctl ctl.c)
 add_executable(binuxjob job.c)
 add_executable(binuxbridge bridge.c)
 foreach(target binuxctl binuxjob binuxbridge)
  set_target_properties(${target} PROPERTIES OUTPUT_NAME "lib${target}.so")
  target_compile_options(${target} PRIVATE -fPIE)
  target_link_options(${target} PRIVATE -pie -Wl,-z,max-page-size=16384)
 endforeach()
else()
 add_executable(binuxctl ctl.c)
 add_executable(binuxjob job.c)
 add_executable(binuxbridge bridge.c)
endif()
